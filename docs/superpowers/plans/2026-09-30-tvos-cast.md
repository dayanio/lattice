# tvOS 投屏实施计划（RefluxAppleTV 内嵌引擎）

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Mac/iPhone 发起投屏，Apple TV（RefluxAppleTV 内嵌 Lattice 引擎）经 overlay 拉流真实播放，补齐苹果生态最后一端。

**Tech Stack:** Go 1.26（`-buildmode=c-archive`，GOOS=ios + appletvos SDK）、SwiftUI（tvOS 17+）、PlayerKit（`../../PlayerKit` 本地包）、XcodeGen、NWListener、CoreImage 二维码。

**Spec:** `docs/superpowers/specs/2026-09-30-tvos-cast-design.md`

```go
// SetEventHandler registers the host callback. Events are plain strings;
// cast commands arrive as "cast: {json}" — the same format the
// tunnel-extension engine emits through its gomobile delegate
// (apple/engine/engine.go:410). Dedup by command id is the host's job.
func (e *EmbeddedEngine) SetEventHandler(fn func(event string)) {
	e.mu.Lock()
	e.eventFn = fn
	e.mu.Unlock()
}

func (e *EmbeddedEngine) emitEvent(event string) {
	e.mu.Lock()
	fn := e.eventFn
	e.mu.Unlock()
	if fn != nil {
		fn(event)
	}
}

// dispatchCastCommand is the CastCommandHandler sink (NATS fallback
// transport). The primary overlay-interception path does not exist in the
// embedded engine; see the tvOS cast design §七.
func (e *EmbeddedEngine) dispatchCastCommand(payload []byte) {
	e.emitEvent("cast: " + string(payload))
}
```

```go
	}

// serveRange: GET 无 Range → 200 全量 + Content-Length；有 Range → 206 分片。
func serveRange(t *testing.T, w http.ResponseWriter, r *http.Request, body []byte) {
	t.Helper()
	spec := r.Header.Get("Range")
	if spec == "" {
		w.Header().Set("Content-Length", strconv.Itoa(len(body)))
		w.Write(body)
		return
	}
	var start int
	if _, err := fmt.Sscanf(spec, "bytes=%d-", &start); err != nil {
		t.Fatalf("bad range %q", spec)
	}
	w.Header().Set("Content-Range", fmt.Sprintf("bytes %d-%d/%d", start, len(body)-1, len(body)))
	w.Header().Set("Content-Length", strconv.Itoa(len(body)-start))
	w.WriteHeader(http.StatusPartialContent)
	w.Write(body[start:])
}
```

```go
// Package tvoslib exports the embedded Lattice engine over a plain C ABI
// for tvOS apps that link the static library directly (no gomobile). See
// docs/superpowers/specs/2026-09-30-tvos-cast-design.md §4.2.
package tvoslib

import (
	"context"
	"fmt"
	"io"
	"net"
	"net/http"
	"strconv"
	"strings"
	"sync"
)

// mediaHandle serves one URL over an injectable dial function. Production
// dials through the embedded engine's netstack; tests dial loopback.
// ReadAt keeps one Range stream open and follows it while the caller reads
// sequentially (the common case); any offset jump reopens with a fresh
// Range. All methods are safe for concurrent use — PlayerKit may call read
// from multiple threads (see MediaRandomAccessReader's contract).
type mediaHandle struct {
	url   string
	dial  func(ctx context.Context, network, addr string) (net.Conn, error)
	size  int64
	client *http.Client

	mu     sync.Mutex
	body   io.ReadCloser
	pos    int64 // next byte the open body will return
	closed bool
}

func newMediaHandle(urlStr string, dial func(ctx context.Context, network, addr string) (net.Conn, error)) (*mediaHandle, error) {
	h := &mediaHandle{
		url:   urlStr,
		dial:  dial,
		size:  -1,
		client: &http.Client{Transport: &http.Transport{
			DialContext:       dial,
			DisableCompression: true,
		}},
	}
	// Probe total size with a 1-byte Range; servers without Range support
	// return 200 + Content-Length, which we also accept (no seek then).
	req, err := http.NewRequest("GET", urlStr, nil)
	if err != nil {
		return nil, err
	}
	req.Header.Set("Range", "bytes=0-0")
	resp, err := h.client.Do(req)
	if err != nil {
		return nil, err
	}
	resp.Body.Close()
	if cr := resp.Header.Get("Content-Range"); cr != "" {
		// "bytes 0-0/12345"
		if i := strings.LastIndex(cr, "/"); i >= 0 {
			if n, err := strconv.ParseInt(cr[i+1:], 10, 64); err == nil {
				h.size = n
			}
		}
	} else if resp.StatusCode == http.StatusOK && resp.ContentLength > 0 {
		h.size = resp.ContentLength
	}
	return h, nil
}

// readAt reads up to len(buf) bytes starting at offset. Returns bytes read
// (0 = EOF) or -1 on error.
func (h *mediaHandle) readAt(offset int64, buf []byte) int {
	h.mu.Lock()
	defer h.mu.Unlock()
	if h.closed {
		return -1
	}
	if h.body == nil || h.pos != offset {
		if err := h.openAt(offset); err != nil {
			return -1
		}
	}
	n, err := io.ReadFull(h.body, buf)
	h.pos += int64(n)
	if err == io.EOF || err == io.ErrUnexpectedEOF {
		return n
	}
	if err != nil {
		h.body.Close()
		h.body = nil
		return -1
	}
	return n
}

func (h *mediaHandle) openAt(offset int64) error {
	if h.body != nil {
		h.body.Close()
		h.body = nil
	}
	req, err := http.NewRequest("GET", h.url, nil)
	if err != nil {
		return err
	}
	req.Header.Set("Range", fmt.Sprintf("bytes=%d-", offset))
	resp, err := h.client.Do(req)
	if err != nil {
		return err
	}
	if resp.StatusCode != http.StatusPartialContent && resp.StatusCode != http.StatusOK {
		resp.Body.Close()
		return fmt.Errorf("range get: status %d", resp.StatusCode)
	}
	h.body = resp.Body
	h.pos = offset
	return nil
}

func (h *mediaHandle) close() {
	h.mu.Lock()
	defer h.mu.Unlock()
	if h.body != nil {
		h.body.Close()
		h.body = nil
	}
	h.closed = true
}
```

```go
/*
#include <stdlib.h>
typedef void (*TVEventFn)(const char* event, void* ctx);
static void tvoslibDispatch(TVEventFn fn, void* ctx, const char* event) {
	if (fn != 0) { fn(event, ctx); }
}
*/
import "C"

import (
	"context"
	"net"
	"sync"
	"unsafe"

	"github.com/alatticeio/lattice/apple/engine/embedded"
)

var (
	mu     sync.Mutex
	engine *embedded.EmbeddedEngine
)

func setEngine(e *embedded.EmbeddedEngine) { mu.Lock(); engine = e; mu.Unlock() }
func getEngine() *embedded.EmbeddedEngine  { mu.Lock(); defer mu.Unlock(); return engine }

//export TVStart
func TVStart(cfg *C.char, onEvent C.TVEventFn, ctx unsafe.Pointer) *C.char {
	e, err := embedded.New(C.GoString(cfg))
	if err != nil {
		return C.CString("config: " + err.Error())
	}
	if onEvent != nil {
		e.SetEventHandler(func(event string) {
			cEvent := C.CString(event)
			defer C.free(unsafe.Pointer(cEvent))
			C.tvoslibDispatch(onEvent, ctx, cEvent)
		})
	}
	if err := e.StartAsync(); err != nil {
		return C.CString("start: " + err.Error())
	}
	setEngine(e)
	return nil
}

//export TVStop
func TVStop() {
	if e := getEngine(); e != nil {
		_ = e.Stop()
	}
	setEngine(nil)
}

//export TVOverlayAddress
func TVOverlayAddress() *C.char {
	e := getEngine()
	if e == nil {
		return C.CString("")
	}
	return C.CString(e.OverlayAddress())
}

//export TVOpenURL
func TVOpenURL(url *C.char, handleOut *unsafe.Pointer) *C.char {
	e := getEngine()
	if e == nil {
		return C.CString("engine not started")
	}
	h, err := newMediaHandle(C.GoString(url), func(ctx context.Context, network, addr string) (net.Conn, error) {
		return e.Dial(ctx, network, addr)
	})
	if err != nil {
		return C.CString("open: " + err.Error())
	}
	*handleOut = unsafe.Pointer(h)
	return nil
}

//export TVTotalSize
func TVTotalSize(handle unsafe.Pointer) C.int64_t {
	return C.int64_t((*mediaHandle)(handle).size)
}

//export TVReadAt
func TVReadAt(handle unsafe.Pointer, offset C.int64_t, length C.int, buf *C.char) C.int {
	h := (*mediaHandle)(handle)
	out := unsafe.Slice((*byte)(unsafe.Pointer(buf)), int(length))
	return C.int(h.readAt(int64(offset), out))
}

//export TVClose
func TVClose(handle unsafe.Pointer) {
	(*mediaHandle)(handle).close()
}

//export TVFree
func TVFree(p *C.char) {
	C.free(unsafe.Pointer(p))
}

func main() {}
```

```swift
import Foundation

/// Swift 侧对 LatticeTVCore.a（tvoslib C ABI）的薄封装。
/// 引擎生命周期 + cast 事件 + 媒体句柄；见 tvOS cast 设计 §4.2/§五。
final class TVEngine: @unchecked Sendable {
    /// 事件即字符串，cast 命令为 "cast: {json}"（与 iOS 委托同格式）。
    private let lock = NSLock()
    private var eventHandler: (@Sendable (String) -> Void)?
    private let box = UnmanagedBox()

    private final class UnmanagedBox: @unchecked Sendable {
        var handler: (@Sendable (String) -> Void)?
    }

    /// 返回 nil 表示已启动（bring-up 异步进行，overlayAddress 轮询）；
    /// 返回错误串（调用方展示/打日志）。
    func start(cfgJSON: String, onEvent: @escaping @Sendable (String) -> Void) -> String? {
        box.handler = onEvent
        let result = cfgJSON.withCString { cfg in
            TVStart(cfg, { event, ctx in
                guard let event, let ctx else { return }
                let box = Unmanaged<UnmanagedBox>.fromOpaque(ctx).takeUnretainedValue()
                box.handler?(String(cString: event))
            }, Unmanaged.passUnretained(box).toOpaque())
        }
        return result == nil ? nil : String(cString: result!)
    }

    func stop() {
        box.handler = nil
        TVStop()
    }

    var overlayAddress: String {
        guard let p = TVOverlayAddress() else { return "" }
        defer { TVFree(p) }
        return String(cString: p)
    }

    func openMedia(url: String) -> TVMediaHandle? {
        var handle: UnsafeMutableRawPointer?
        let err = url.withCString { TVOpenURL($0, &handle) }
        if err != nil {
            let message = String(cString: err)
            TVFree(err)
            NSLog("[latticetv] openMedia failed: \(message)")
            return nil
        }
        guard let handle else { return nil }
        return CMediaHandle(handle)
    }

    func free(_ p: UnsafeMutablePointer<CChar>) { TVFree(p) }
}

/// TVEngine.openMedia 的返回：MediaRandomAccessReader 的字节源。
protocol TVMediaHandle: AnyObject, Sendable {
    var byteCount: Int64 { get }
    func readAt(offset: Int64, length: Int, into buffer: UnsafeMutableRawBufferPointer) -> Int
    func closeHandle()
}

final class CMediaHandle: TVMediaHandle {
    let raw: UnsafeMutableRawPointer
    init(_ raw: UnsafeMutableRawPointer) { self.raw = raw }
    var byteCount: Int64 { Int64(TVTotalSize(raw)) }
    func readAt(offset: Int64, length: Int, into buffer: UnsafeMutableRawBufferPointer) -> Int {
        Int(TVReadAt(raw, offset, Int32(length), buffer.baseAddress.map { $0.assumingMemoryBound(to: CChar.self) }))
    }
    func closeHandle() { TVClose(raw) }
    deinit { TVClose(raw) }
}
```

```swift
import Foundation
import PlayerKit

/// MediaRandomAccessReader 的 overlay 实现：字节全部来自引擎句柄
/// （经 netstack 拨号到媒体源的 Range 连接）。满足协议的线程契约：
/// read 是同步调用、本类自身无共享可变状态（句柄内部保证线程安全）。
final class OverlayMediaReader: MediaRandomAccessReader, @unchecked Sendable {
    private let handle: TVMediaHandle
    private let lock = NSLock()
    private var isClosed = false

    init(handle: TVMediaHandle) { self.handle = handle }

    var totalSize: Int64 { handle.byteCount }

    func read(offset: Int64, length: Int, into buffer: UnsafeMutableRawBufferPointer) throws -> Int {
        lock.lock()
        defer { lock.unlock() }
        guard !isClosed else { throw CastReaderError.closed }
        let n = handle.readAt(offset: offset, length: length, into: buffer)
        guard n >= 0 else { throw CastReaderError.readFailed }
        return n
    }

    func close() {
        lock.lock()
        defer { lock.unlock() }
        guard !isClosed else { return }
        isClosed = true
        handle.closeHandle()
    }

    var downloadedUpToOffset: Int64 { -1 }
}

enum CastReaderError: Error {
    case closed
    case readFailed
}
```

```swift
/// 入网配置的 Keychain 持久化。privateKey 必须持久化复用——
/// 同名设备换 WG 私钥重注册会被管理面拒绝。
struct TVCastConfig: Codable, Equatable {

    var token: String
    var name: String
    var privateKey: String?

enum TVCastConfigStore {
    private static let key = "latticetv.config"

    static func load(keychain: KeychainStore) -> TVCastConfig? {
        guard let raw = keychain.get(key), let data = raw.data(using: .utf8) else { return nil }
        return try? JSONDecoder().decode(TVCastConfig.self, from: data)

    static func save(_ config: TVCastConfig, keychain: KeychainStore) {
        guard let data = try? JSONEncoder().encode(config),
              let raw = String(data: data, encoding: .utf8) else { return }
        keychain.set(raw, key: key)
    }

    static func clear(keychain: KeychainStore) {
        keychain.delete(key)
    }
}
```

```swift
import Foundation
import OSLog
import UIKit

private let logger = Logger(subsystem: "io.reflux.apple", category: "TVCastManager")

/// 电视端投屏管理器：引擎生命周期 + cast 命令 → 播放。
/// 命令通道 v1 = NATS 兜底（引擎内订阅），格式 "cast: {json}"，
/// 按 id 去重（保留 32 个）。
@MainActor
@Observable
final class TVCastManager {
    enum State: Equatable {
        case notEnrolled
        case joining
        case online(String)   // overlay IP
        case playing(String)  // 标题
    }

    private(set) var state: State = .notEnrolled
    private(set) var lastError: String?

    private let engine = TVEngine()
    private let keychain: KeychainStore
    private weak var playerController: PlayerController?
    private var recentCommandIDs: [String] = []
    private var pollTask: Task<Void, Never>?

    init(keychain: KeychainStore, playerController: PlayerController?) {
        self.keychain = keychain
        self.playerController = playerController
    }

    func startIfNeeded() {
        guard case .notEnrolled = state, let cfg = TVCastConfigStore.load(keychain: keychain) else { return }
        startEngine(config: cfg)
    }

    /// 粘贴 join payload 入网（v1 占位；Phase 3 换扫码配对）。
    /// 接受 `lattice://join?server=&token=&name=` 或裸 token（server 用现有 defaults）。
    func enroll(payloadText: String) -> String? {
        guard let server = URL(string: payloadText),
              server.scheme == "lattice",
              let items = URLComponents(url: server, resolvingAgainstBaseLocation: false)?.queryItems,
              let token = items.first(where: { $0.name == "token" })?.value else {
            return "无法解析入网配置"
        }
        let host = items.first(where: { $0.name == "server" })?.value ?? ""
        let name = items.first(where: { $0.name == "name" })?.value ?? "Apple-TV"
        let cfg = TVCastConfig(serverURL: host, token: token, name: name, privateKey: nil)
        TVCastConfigStore.save(cfg, keychain: keychain)
        startEngine(config: cfg)
        return nil
    }

    private func startEngine(config: TVCastConfig) {
        state = .joining
        var cfgJSON = "{\"serverURL\":\"\(config.serverURL)\",\"token\":\"\(config.token)\",\"name\":\"\(config.name)\""
        if let pk = config.privateKey {
            cfgJSON += ",\"privateKey\":\"\(pk)\""
        }
        cfgJSON += "}"
        let err = engine.start(cfgJSON: cfgJSON) { [weak self] event in
            Task { @MainActor in self?.handleEvent(event) }
        }
        if let err {
            lastError = err
            state = .notEnrolled
            return
        }
        pollTask = Task { [weak self] in
            while !Task.isCancelled {
                if let self, let addr = self.pollOverlayAddress() { break }
                try? await Task.sleep(nanoseconds: 500_000_000)
            }
        }
    }

    private func pollOverlayAddress() -> String? {
        let addr = engine.overlayAddress
        guard !addr.isEmpty else { return nil }
        state = .online(addr)
        UIApplication.shared.isIdleTimerDisabled = true
        persistPrivateKey()
        return addr
    }

    /// 引擎生成的 WG 私钥要立即回存（重注册换 key 被拒）。
    private func persistPrivateKey() { /* Task 7b: 经 tvoslib 暴露 PrivateKey 后回存；首版留空注释占位 */ }

    private func handleEvent(_ event: String) {
        guard event.hasPrefix("cast: ") else { return }
        let payload = String(event.dropFirst("cast: ".count))
        guard let data = payload.data(using: .utf8),
              let cmd = try? JSONDecoder().decode(CastCommand.self, from: data) else {
            logger.error("cast event unparsable: \(event.prefix(200), privacy: .public)")
            return
        }
        guard dedup(cmd.id) else { return }
        guard cmd.action == "play", let url = URL(string: cmd.url) else { return }
        play(url: url, title: cmd.title)
    }

    private func play(url: URL, title: String) {
        guard let playerController else { return }
        guard let handle = engine.openMedia(url: url.absoluteString) else {
            lastError = "源连接失败"
            return
        }
        state = .playing(title)
        playerController.playReader(OverlayMediaReader(handle: handle))
    }

    private func dedup(_ id: String) -> Bool {
        if recentCommandIDs.contains(id) { return false }
        recentCommandIDs.append(id)
        if recentCommandIDs.count > 32 { recentCommandIDs.removeFirst(recentCommandIDs.count - 32) }
        return true
    }
}

private struct CastCommand: Decodable {
    let id: String
    let action: String
    let url: String
    let title: String?
}
```

```swift
import Foundation
import Network

/// 局域网一次性配对监听：电视亮码，手机扫码后把入网配置 POST 过来。
/// 只收一次配置即关；code 不匹配一律 403（防同网误投/恶意下发）。
@MainActor
final class TVPairingService {
    struct PairingOffer: Equatable {
        let lanURL: String
        let code: String
        let name: String
    }

    private var listener: NWListener?
    private var onConfig: ((TVCastConfig) -> Void)?

    func begin(name: String, onConfig: @escaping (TVCastConfig) -> Void) throws -> PairingOffer {
        self.onConfig = onConfig
        let code = String(format: "%04d", Int.random(in: 0...9999))
        let params = NWParameters.tcp
        params.allowLocalEndpointReuse = true
        let listener = try NWListener(using: params)
        self.listener = listener

        let offer = PairingOffer(lanURL: "http://\(Self.lanIPv4() ?? "0.0.0.0"):<port>", code: code, name: name)
        // port 在 listener 启动后才能确定：stateUpdateHandler 里回填 offer.lanURL。

        listener.newConnectionHandler = { [weak self] conn in
            self?.handle(conn, code: code, name: name)
        }
        listener.stateUpdateHandler = { state in
            if case .ready = state, let port = listener.port {
                NotificationCenter.default.post(name: .tvPairingReady, object: PairingOffer(
                    lanURL: "http://\(Self.lanIPv4() ?? "0.0.0.0"):\(port)", code: code, name: name))
            }
        }
        listener.start(queue: .global(qos: .userInitiated))
        return offer
    }

    func stop() {
        listener?.cancel()
        listener = nil
    }

    // 最小 HTTP 解析：读请求头 → 校验 code → 读 body → 回 200/403 → 关连接。
    private func handle(_ conn: NWConnection, code: String, name: String) {
        conn.start(queue: .global(qos: .userInitiated))
        receiveRequest(conn) { [weak self] request in
            guard let request,
                  request.method == "POST",
                  request.path == "/config",
                  request.headers["X-Pair-Code"] == code,
                  let data = request.body.data(using: .utf8),
                  let cfg = try? JSONDecoder().decode(TVCastConfig.self, from: data) else {
                Self.respond(conn, "HTTP/1.1 403 Forbidden\r\nContent-Length: 0\r\n\r\n")
                return
            }
            Self.respond(conn, "HTTP/1.1 200 OK\r\nContent-Length: 2\r\n\r\nok")
            Task { @MainActor in
                self?.stop()
                self?.onConfig?(cfg)
            }
        }
    }

    private func receiveRequest(_ conn: NWConnection, completion: @escaping (HTTPRequest?) -> Void) {
        var buffer = Data()
        func recv() {
            conn.receive(minimumIncompleteLength: 1, maximumLength: 64 * 1024) { data, _, _, error in
                if let data { buffer.append(data) }
                if let request = HTTPRequest.parse(buffer) { completion(request); return }
                if error != nil || buffer.count > 1024 * 1024 { completion(nil); return }
                recv()
            }
        }
        recv()
    }

    static func respond(_ conn: NWConnection, _ text: String) {
        conn.send(content: Data(text.utf8), completion: .contentProcessed { _ in
            conn.cancel()
        })
    }

    static func lanIPv4() -> String? {
        var address: String?
        var ifaddr: UnsafeMutablePointer<ifaddrs>?
        guard getifaddrs(&ifaddr) == 0 else { return nil }
        defer { freeifaddrs(ifaddr) }
        var ptr = ifaddr
        while let p = ptr {
            let iface = p.pointee
            if let sa = iface.ifa_addr, sa.pointee.sa_family == UInt8(AF_INET) {
                let name = String(cString: iface.ifa_name)
                if name == "en0" {
                    var host = [CChar](repeating: 0, count: Int(NI_MAXHOST))
                    if getnameinfo(sa, socklen_t(sa.pointee.sa_len), &host, socklen_t(host.count), nil, 0, NI_NUMERICHOST) == 0 {
                        address = String(cString: host)
                    }
                }
            }
            ptr = iface.ifa_next
        }
        return address
    }
}

extension Notification.Name { static let tvPairingReady = Notification.Name("tvPairingReady") }

struct HTTPRequest {
    let method: String
    let path: String
    let headers: [String: String]
    let body: String

    static func parse(_ data: Data) -> HTTPRequest? {
        guard let text = String(data: data, encoding: .utf8),
              let headEnd = text.range(of: "\r\n\r\n") else { return nil }
        let head = String(text[..<headEnd.lowerBound])
        let body = String(text[headEnd.upperBound...])
        var lines = head.components(separatedBy: "\r\n")
        guard !lines.isEmpty else { return nil }
        let parts = lines.removeFirst().split(separator: " ")
        guard parts.count >= 2 else { return nil }
        var headers: [String: String] = [:]
        for line in lines {
            if let i = line.firstIndex(of: ":") {
                headers[String(line[..<i]).lowercased()] = line[line.index(after: i)...].trimmingCharacters(in: .whitespaces)
            }
        }
        return HTTPRequest(method: String(parts[0]), path: String(parts[1]), headers: headers, body: body)
    }
}
```
